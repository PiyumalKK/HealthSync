\section{Segment 1: Homepage \& Public Pages \hfill \textcolor{timecolor}{\normalsize 0:00 -- 0:40}}

\textbf{Open the app in browser.} Start at the homepage.

\begin{enumerate}[leftmargin=1.5em, label=\textbf{\arabic*.}]
  \item \textbf{Hero Section} --- Scroll briefly. Show the hero banner with \textit{``Book Your Doctor''} call-to-action and the live statistics (doctors, patients, appointments count).
  \item \textbf{Featured Doctors} --- Scroll down to the featured doctors carousel. Point out doctor cards with ratings, specialization, and consultation fees in LKR.
  \item \textbf{Services Page} --- Click \textbf{Services} in the navbar. Briefly show the 6 service cards (Find Doctors, Book Appointments, Prescriptions, Notifications, Health Records, Security).
  \item \textbf{About Page} --- Click \textbf{About}. Flash the mission statement and team section.
\end{enumerate}

\textbf{Say:} \textit{``This is HealthSync, a cloud-native healthcare platform built with 7 microservices on Azure. Let me show you the key user flows.''}

%-------------------------------------------------------------
\section{Segment 2: Patient Registration \& Login \hfill \textcolor{timecolor}{\normalsize 0:40 -- 1:10}}

\begin{enumerate}[leftmargin=1.5em, label=\textbf{\arabic*.}]
  \item Click \textbf{Get Started} or \textbf{Sign In} in the navbar.
  \item \textbf{Register Page} --- Show the registration form (First Name, Last Name, Email, Password, Confirm Password). Point out the Google OAuth button.
  \item \textbf{Google Sign-In} --- Click \textit{``Sign in with Google''} to demonstrate OAuth 2.0 flow. \textit{(Or register with email if preferred.)}
  \item Once logged in, you land on the \textbf{Patient Dashboard} automatically.
\end{enumerate}

\textbf{Say:} \textit{``Users can register with email or Google OAuth. Authentication uses JWT tokens managed through our API Gateway.''}

%-------------------------------------------------------------
\section{Segment 3: Doctor Browsing \& Booking \hfill \textcolor{timecolor}{\normalsize 1:10 -- 1:50}}

\begin{enumerate}[leftmargin=1.5em, label=\textbf{\arabic*.}]
  \item Click \textbf{Find Doctors} in the navbar.
  \item \textbf{Search \& Filter} --- Type a name in the search bar. Change the specialization dropdown (e.g., Cardiology). Show results updating.
  \item \textbf{Doctor Detail} --- Click on a doctor card. Show the full profile: qualifications, experience, bio, languages, patient reviews, and consultation fee.
  \item \textbf{Book Appointment} --- Select a date and time slot. Click \textbf{``Pay Now \& Book''} to show Stripe checkout integration. \textit{(Or click ``Confirm Booking'' for pay-later flow.)}
  \item Show the success confirmation / redirect to Appointments page.
\end{enumerate}

\textbf{Say:} \textit{``The Doctor Service provides search and filtering. Appointments go through the Appointment Service, and payments are handled via Stripe integration.''}

%-------------------------------------------------------------
\section{Segment 4: Dashboard \& Notifications \hfill \textcolor{timecolor}{\normalsize 1:50 -- 2:20}}

\begin{enumerate}[leftmargin=1.5em, label=\textbf{\arabic*.}]
  \item Navigate to \textbf{Dashboard}. Show the patient dashboard with upcoming appointments, recent prescriptions, and health summary cards.
  \item Click \textbf{Appointments} in the navbar. Show the appointments list with status badges (Pending, Confirmed, Completed, Cancelled).
  \item Click the \textbf{bell icon} or navigate to \textbf{Notifications}. Show real-time notifications (appointment confirmations, prescription alerts).
  \item Briefly click \textbf{Profile} to show the profile management page.
\end{enumerate}

\textbf{Say:} \textit{``Patients get a unified dashboard. The Notification Service runs on Python FastAPI and pushes real-time alerts via Azure Service Bus.''}

%-------------------------------------------------------------
\section{Segment 5: Doctor Portal \& Prescriptions \hfill \textcolor{timecolor}{\normalsize 2:20 -- 2:50}}

\textit{(Log in as a doctor account, or switch accounts.)}

\begin{enumerate}[leftmargin=1.5em, label=\textbf{\arabic*.}]
  \item Show the \textbf{Doctor Dashboard} with today's appointments, patient queue, and quick stats.
  \item Click on a completed appointment. Click \textbf{``Write Prescription''} to open the prescription form.
  \item \textbf{Prescription Form} --- Show adding medications (name, dosage, frequency, duration), diagnosis, and notes. Submit.
  \item Navigate to \textbf{My Schedule} (Doctor Availability page). Show setting available time slots.
\end{enumerate}

\textbf{Say:} \textit{``Doctors manage appointments and write digital prescriptions. The Prescription Service stores records securely, and patients receive instant notifications.''}

\section*{Pre-Demo Checklist}

\begin{itemize}[leftmargin=1.5em]
  \item[$\square$] App is live and accessible at the Front Door URL
  \item[$\square$] Have a patient test account ready (or register live)
  \item[$\square$] Have a doctor test account ready (pre-verified by admin)
  \item[$\square$] At least 1 existing appointment in ``Confirmed'' status for prescription demo
  \item[$\square$] Browser zoom set to 90--100\%, bookmarks bar hidden
  \item[$\square$] Screen recorder running (OBS / Windows Game Bar / browser extension)
  \item[$\square$] Microphone tested
\end{itemize}

\vspace{8pt}

\section*{Recording Tips}

\begin{itemize}[leftmargin=1.5em]
  \item Use \textbf{1920$\times$1080} resolution for crisp recording.
  \item \textbf{Slow your mouse} --- move deliberately so viewers can follow.
  \item \textbf{Pause 1--2 seconds} on each page before clicking the next thing.
  \item If something loads slowly, fill time with narration about the architecture.
  \item Record in one take if possible; edit out long pauses afterwards.
\end{itemize}
